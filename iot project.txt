// ======================================

#define MQ2_SENSOR A0
#define YELLOW_LED D1
#define RED_LED D2
#define BUZZER D5

// ======================================
// SENSOR CONFIGURATION
// ======================================
const int GAS_THRESHOLD = 240;
const int RESET_THRESHOLD = 220;

const unsigned long WARMUP_TIME = 60000UL;
const unsigned long READ_INTERVAL = 500UL;

// ======================================
// OUTPUT POLARITY
// Change to true only if that output
// is wired active-LOW.
// ======================================
const bool YELLOW_ACTIVE_LOW = false;
const bool RED_ACTIVE_LOW = false;
const bool BUZZER_ACTIVE_LOW = false;

// ======================================
// WEB SERVER
// ======================================
ESP8266WebServer server(80);

// ======================================
// GLOBAL VARIABLES
// ======================================
int gasValue = 0;

bool sensorReady = false;
bool smokeDetected = false;

unsigned long startupTime = 0;
unsigned long lastRead = 0;
unsigned long lastWiFiCheck = 0;

// ======================================
// OUTPUT FUNCTION
// ======================================
void writeOutput(int pin, bool turnOn, bool activeLow) {
  digitalWrite(
    pin,
    turnOn ? (activeLow ? LOW : HIGH)
           : (activeLow ? HIGH : LOW)
  );
}

// ======================================
// ALARM CONTROL
// ======================================
void updateAlarm() {

  bool normalMode = sensorReady && !smokeDetected;
  bool dangerMode = sensorReady && smokeDetected;

  writeOutput(YELLOW_LED, normalMode, YELLOW_ACTIVE_LOW);
  writeOutput(RED_LED, dangerMode, RED_ACTIVE_LOW);
  writeOutput(BUZZER, dangerMode, BUZZER_ACTIVE_LOW);
}

// ======================================
// SENSOR READING
// ======================================
void readMQ2() {

  long total = 0;

  for (int i = 0; i < 10; i++) {
    total += analogRead(MQ2_SENSOR);
    delay(2);
  }

  gasValue = total / 10;

  if (millis() - startupTime < WARMUP_TIME) {

    sensorReady = false;
    smokeDetected = false;

  } else {

    sensorReady = true;

    if (gasValue >= GAS_THRESHOLD) {
      smokeDetected = true;
    }

    if (gasValue <= RESET_THRESHOLD) {
      smokeDetected = false;
    }
  }

  updateAlarm();

  // SERIAL MONITOR
  Serial.println("--------------------------");

  Serial.print("MQ2 Reading: ");
  Serial.println(gasValue);

  Serial.print("Sensor Ready: ");
  Serial.println(sensorReady ? "YES" : "NO");

  Serial.print("Smoke Status: ");
  Serial.println(smokeDetected ? "DETECTED" : "NORMAL");

  Serial.print("Yellow D1 GPIO: ");
  Serial.println(digitalRead(YELLOW_LED));

  Serial.print("Red D2 GPIO: ");
  Serial.println(digitalRead(RED_LED));

  Serial.print("Buzzer D5 GPIO: ");
  Serial.println(digitalRead(BUZZER));

  Serial.println("--------------------------");
}

// ======================================
// DASHBOARD ROUTE
// ======================================
void handleDashboard() {
  server.sendHeader("Cache-Control","no-store");
  server.send_P(200,"text/html",DASHBOARD);
}

// ======================================
// SENSOR DATA ROUTE
// ======================================
void handleData() {

  String json = "{";

  json += "\"gas\":" + String(gasValue);

  json += ",\"sensor_ready\":";
  json += sensorReady ? "true" : "false";

  json += ",\"alarm\":";
  json += smokeDetected ? "true" : "false";

  json += ",\"wifi\":";
  json += WiFi.status() == WL_CONNECTED ? "true" : "false";

  json += "}";

  server.sendHeader("Cache-Control","no-store");
  server.send(200,"application/json",json);
}

// ======================================
// SETUP
// ======================================
void setup() {

  Serial.begin(115200);

  pinMode(YELLOW_LED,OUTPUT);
  pinMode(RED_LED,OUTPUT);
  pinMode(BUZZER,OUTPUT);

  startupTime = millis();

  updateAlarm();

  Serial.println();
  Serial.println("INNOVEX MQ-2 MONITORING SYSTEM");

  WiFi.mode(WIFI_STA);
  WiFi.begin(WIFI_SSID,WIFI_PASSWORD);

  Serial.println("Connecting Wi-Fi...");

  unsigned long start = millis();

  while(WiFi.status() != WL_CONNECTED &&
        millis()-start < 20000UL) {

    delay(500);
    Serial.print(".");
  }

  Serial.println();

  if(WiFi.status() == WL_CONNECTED) {

    Serial.println("Wi-Fi Connected!");

    Serial.print("Dashboard IP: http://");
    Serial.println(WiFi.localIP());

  } else {

    Serial.println("Wi-Fi connection pending");

  }

  server.on("/",HTTP_GET,handleDashboard);
  server.on("/data",HTTP_GET,handleData);

  server.begin();

  Serial.println("Web server started");
  Serial.println("60-second sensor warm-up");
}

// ======================================
// MAIN LOOP
// ======================================
void loop() {

  server.handleClient();

  if(WiFi.status() != WL_CONNECTED &&
     millis()-lastWiFiCheck >= 10000UL) {

    lastWiFiCheck = millis();
    WiFi.begin(WIFI_SSID,WIFI_PASSWORD);
  }

  if(millis()-lastRead >= READ_INTERVAL) {

    lastRead = millis();
    readMQ2();
  }
}
